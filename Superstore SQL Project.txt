--Step 1: Create Database
CREATE DATABASE superstore_db;

--Step 3: Create Customers Table
CREATE TABLE customers(
  customer_id VARCHAR(20)
PRIMARY KEY,
  customer_name VARCHAR(100) ,
  segment VARCHAR(50) ,
  country VARCHAR(50) ,
  city VARCHAR(50) ,
  state VARCHAR(50) ,
  postal_code INT ,
  region VARCHAR(50) 
);

--Explanation: This step creates the customers table to store customer information
such as customer ID, customer name, segment, country, city, state, postal code and region.

--Step 4: Create Products Table
CREATE TABLE products(
  product_id VARCHAR(30)
PRIMARY KEY,
  product_name VARCHAR(200) ,
  category VARCHAR(50), 
  sub_category VARCHAR(50)
  );

--Explanation: This step creates the products table to store product details
including Product ID, category, sub-category, and product name.

--Step 5: Create Orders Table
CREATE TABLE orders(
row_id INT PRIMARY KEY,
order_id VARCHAR(30),
order_date DATE,
ship_date DATE,
ship_mode VARCHAR(30),
customer_id VARCHAR(20),
product_id VARCHAR(30),
sales DECIMAL(10,2),
quantity INT,
discount DECIMAL(5,2),
profit DECIMAL(10,2),
order_year INT,
order_month VARCHAR(20),
order_quarter VARCHAR(10),
order_day INT,
shipping_days INT,

FOREIGN KEY (customer_id)
    REFERENCES 
 customers(customer_id),
 
FOREIGN KEY (product_id)
     REFERENCES 
products(product_id)
);

--Explanation: This step creates the orders table to store transaction details and
establishes relationships with the customers and products tables using foreign keys.

--products.csv -> products
--orders.csv -> orders

--Explanation: This step imports the cleaned customers.csv, products.csv, and
orders.csv files into their respective PostgreSQL tables for analysis.
